#include "libft.h"

static unsigned int	in_set(char c, char const *set)
{
	unsigned int	j;

	j = 0;
	while (set[j])
	{
		if (c == set[j])
			return (1);
		j++;
	}
	return (0);
}

static size_t	count_set(char const *str, char const *set)
{
	size_t	count;
	size_t	i;

	i = 0;
	count = 0;
	while (str[i])
	{
		if (in_set(str[i], set))
			count++;
		i++;
	}
	return (count);
}

char	*ft_strtrim(char const *s1, char const *set)
{
	char	*trimed;
	size_t	trimed_i;
	size_t	trimed_len;
	size_t	i;

	if (!s1 || !set)
		return (NULL);
	trimed_len = ft_strlen(s1) - count_set(s1, set) + 1;
	trimed = (char *)malloc((trimed_len) * sizeof(char));
	if (!trimed)
		return (NULL);
	i = 0;
	trimed_i = 0;
	while (s1[i])
	{
		if (!in_set(s1[i], set))
		{
			trimed[trimed_i] = s1[i];
			trimed_i++;
		}
		i++;
	}
	trimed[trimed_i] = '\0';
	return (trimed);
}
